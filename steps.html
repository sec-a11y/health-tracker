// 🟢 เพิ่มฟังก์ชันนี้สำหรับดึงวันที่ที่เคยส่งแล้วของพนักงานแต่ละคน function doGet(e) { // รองรับส่วนอนุมัติเดิม if (e.parameter.action === 'approve') { return handleApproval(e); } // รองรับการดึงข้อมูลแดชบอร์ดเดิม if (e.parameter.action === 'get_leaderboard') { return handleGetLeaderboard(e); } // 👇 เพิ่มส่วนนี้: ดึงรายการวันที่ที่เคยส่งแล้วของ User คนนี้ if (e.parameter.action === 'get_submitted_dates') { let uid = e.parameter.uid; let sheet = SpreadsheetApp.openById(SPREADSHEET_ID); let ws = sheet.getSheetByName("Daily_Steps"); let data = ws.getDataRange().getValues(); let submittedDates = []; for (let i = 1; i < data.length; i++) { let rowUid = data[i][1]; let rowDate = data[i][2]; // รูปแบบในชีต เช่น 15/09/2026 หรือ YYYY-MM-DD let status = data[i][5]; // "รออนุมัติ" หรือ "อนุมัติ" if (rowUid === uid && (status === "รออนุมัติ" || status === "อนุมัติ")) { // แปลงวันที่ในชีตให้เป็นรูปแบบ YYYY-MM-DD เพื่อให้ตรงกับ input type="date" let parts = rowDate.split('/'); if (parts.length === 3) { let formattedDate = `${parts[2]}-${parts[1]}-${parts[0]}`; submittedDates.push(formattedDate); } else { submittedDates.push(rowDate); } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", submittedDates: submittedDates })).setMimeType(ContentService.MimeType.JSON); } }
